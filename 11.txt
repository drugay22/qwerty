class PartnerApp(tkinter.Tk):

        super().__init__()
        self.title("РЎРїРёСЃРѕРє РїР°СЂС‚РЅРµСЂРѕРІ")
        self.geometry("800x600")
        self.protocol("WM_DELETE_WINDOW", self.on_closing)
        # self.iconphoto(False, tkinter.PhotoImage(file=''))
        self.current_frame = None
        self.show_main_page()

    def show_main_page(self):
        if self.current_frame:
            self.current_frame.destroy()
        self.current_frame = MainPage(self)
        self.current_frame.configure(bg='#FFFFFF')
        self.current_frame.pack(fill='both', expand=True)

    def show_edit_partner(self, partner=None):
        if self.current_frame:
            self.current_frame.destroy()
        self.current_frame = PartnerForm(self, partner)
        self.current_frame.configure(bg='#FFFFFF')
        self.current_frame.pack(fill='both', expand=True)

    def on_closing(self):
        if messagebox.askokcancel("Р—Р°РєСЂС‹С‚СЊ РѕРєРЅРѕ", "РҐРѕС‚РёС‚Рµ Р·Р°РєРѕРЅС‡РёС‚СЊ?"):
